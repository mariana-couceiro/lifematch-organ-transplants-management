package com.lifematch.service; import java.time.LocalDate; import java.time.Period; import java.util.List; import com.lifematch.model.Candidate; import com.lifematch.repository.CandidateRepository; public class CandidateService { private final CandidateRepository repository; public CandidateService() { this.repository = new CandidateRepository(); } public Candidate createCandidate( Candidate candidate) { // New candidates always start // on the waiting list candidate.setStatus( "WAITING" ); // Validate information first validateCandidate( candidate ); // Priority is calculated // automatically by the system candidate.setPriority( calculatePriority( candidate ) ); return repository.save( candidate ); } public List getAllCandidates() { return repository.findAll(); } public Candidate getCandidateById( int id) { return repository.findById( id ); } public List getCandidatesByName( String name) { return repository.findByName( name ); } public Candidate updateCandidate( int id, Candidate candidate) { Candidate existingCandidate = repository.findById(id); if (existingCandidate == null) { return null; } /* * The status is preserved. * * The user does not decide whether * a candidate is WAITING or MATCHED * through the edit form. */ candidate.setStatus( existingCandidate.getStatus() ); // Validate updated information validateCandidate( candidate ); /* * Recalculate priority because * birth date, required organ or * urgency may have changed. */ candidate.setPriority( calculatePriority( candidate ) ); return repository.update( id, candidate ); } public boolean deleteCandidate( int id) { return repository.delete( id ); } private int calculatePriority( Candidate candidate) { int score = 0; int age = Period.between( candidate.getBirthDate(), LocalDate.now() ).getYears(); if (age <= 17) { score += 3; } else if (age <= 39) { score += 2; } else if (age <= 64) { score += 1; } switch ( candidate.getRequiredOrgan() ) { case "Heart": case "Liver": case "Lung": score += 3; break; case "Kidney": case "Pancreas": case "Small Intestine": score += 2; break; } switch ( candidate.getUrgency() ) { case "MODERATE": score += 1; break; case "HIGH": score += 2; break; case "CRITICAL": score += 3; break; case "LOW": // LOW adds no points break; } if (score >= 7) { return 4; } if (score >= 5) { return 3; } if (score >= 3) { return 2; } return 1; } private void validateCandidate( Candidate candidate) { if (candidate == null) { throw new IllegalArgumentException( "Candidate data is required." ); } if (candidate.getName() == null || candidate.getName().isBlank()) { throw new IllegalArgumentException( "Candidate name is required." ); } if (candidate.getBirthDate() == null) { throw new IllegalArgumentException( "Birth date is required." ); } if (candidate .getBirthDate() .isAfter( LocalDate.now() )) { throw new IllegalArgumentException( "Birth date cannot be in the future." ); } if (candidate.getBloodType() == null || candidate.getBloodType() .isBlank()) { throw new IllegalArgumentException( "Blood type is required." ); } if (!isValidBloodType( candidate.getBloodType() )) { throw new IllegalArgumentException( "Invalid blood type." ); } if (candidate.getRequiredOrgan() == null || candidate.getRequiredOrgan() .isBlank()) { throw new IllegalArgumentException( "Required organ is required." ); } if (!isValidOrgan( candidate.getRequiredOrgan() )) { throw new IllegalArgumentException( "Invalid required organ." ); } if (candidate.getUrgency() == null || candidate.getUrgency() .isBlank()) { throw new IllegalArgumentException( "Urgency is required." ); } if (!isValidUrgency( candidate.getUrgency() )) { throw new IllegalArgumentException( "Invalid urgency." ); } if (candidate .getWaitingListEntryDate() == null) { throw new IllegalArgumentException( "Waiting list entry date is required." ); } if (candidate .getWaitingListEntryDate() .isAfter( LocalDate.now() )) { throw new IllegalArgumentException( "Waiting list entry date cannot be in the future." ); } if (candidate.getStatus() == null || candidate.getStatus() .isBlank()) { throw new IllegalArgumentException( "Candidate status is required." ); } if (!isValidStatus( candidate.getStatus() )) { throw new IllegalArgumentException( "Invalid candidate status." ); } } private boolean isValidBloodType( String bloodType) { return bloodType.equals("A+") || bloodType.equals("A-") || bloodType.equals("B+") || bloodType.equals("B-") || bloodType.equals("AB+") || bloodType.equals("AB-") || bloodType.equals("0+") || bloodType.equals("0-"); } private boolean isValidOrgan( String organ) { return organ.equals("Kidney") || organ.equals("Liver") || organ.equals("Heart") || organ.equals("Lung") || organ.equals("Pancreas") || organ.equals( "Small Intestine" ); } private boolean isValidUrgency( String urgency) { return urgency.equals("LOW") || urgency.equals("MODERATE") || urgency.equals("HIGH") || urgency.equals("CRITICAL"); } private boolean isValidStatus( String status) { return status.equals("WAITING") || status.equals("MATCHED") || status.equals("INACTIVE"); } }
